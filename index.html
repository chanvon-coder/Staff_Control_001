                1. បើក Google Sheet របស់អ្នក -> ចុចម៉ឺនុយ Extensions (ផ្នែកបន្ថែម) -> ជ្រើសរើស Apps Script។

                3. ចុចប៊ូតុង Deploy (ដាក់ឱ្យដំណើរការ) -> New deployment -> ជ្រើសរើស Web app -> កំណត់ Who has access: Anyone (អ្នកណាក៏បាន)។
                4. ចម្លង Web app URL យកមកបិទភ្ជាប់ក្នុងប្រអប់ «Webhook URL» ក្នុងប្រព័ន្ធនេះ ជាការស្រេច!

                /**
 * Staff Control System - Google Sheets Auto-Transfer Webhook

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Staff_Data") || ss.getActiveSheet();
    
    var headers = [
      "ល.រ", "អត្តលេខ អពដ", "អត្តលេខ កសហវ", "ឈ្មោះឡាតាំង", "ឈ្មោះខ្មែរ",
      "អង្គភាព", "ការិយាល័យ", "តួនាទី", "ភេទ", "ថ្ងៃខែឆ្នាំកំណើត",
      "ថ្ងៃខែឆ្នាំបម្រើការងារ", "ថ្ងៃខែឆ្នាំស្នើសុំ", "ថ្ងៃខែឆ្នាំបញ្ចប់", "ថ្ងៃខែឆ្នាំចាប់ផ្តើម",
      "ប្រចាំឆ្នាំ", "មូលហេតុនៃសំណើ", "ប្រកាសលេខ", "ពិព័ណនាផ្សេងៗ", "ថ្ងៃខែបិទប្រព័ន្ធ",
      "ឯកសារយោង", "ថ្ងៃខែទទួលឯកសារ ឬប្រកាសផ្សេងៗ", "Remark", "ស្ថានភាព", "កាលបរិច្ឆេទធ្វើបច្ចុប្បន្នភាព"
    ];
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(headers);
    }
    
    if (data.action === "SYNC_ALL" && Array.isArray(data.records)) {
      sheet.clearContents();
      sheet.appendRow(headers);
      data.records.forEach(function(r) {
        sheet.appendRow(recordToArray(r));
      });
      return ContentService.createTextOutput(JSON.stringify({ status: "success", count: data.records.length }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    if (data.action === "SYNC_RECORD" && data.record) {
      var r = data.record;
      var values = sheet.getDataRange().getValues();
      var foundRow = -1;
      
      for (var i = 1; i < values.length; i++) {
        if (String(values[i][1]).trim() === String(r.staffId).trim() || String(values[i][0]).trim() === String(r.no).trim()) {
          foundRow = i + 1;
          break;
        }
      }
      
      var rowData = recordToArray(r);
      if (data.syncAction === "DELETE" && foundRow > 1) {
        sheet.deleteRow(foundRow);
      } else if (foundRow > 1) {
        sheet.getRange(foundRow, 1, 1, rowData.length).setValues([rowData]);
      } else {
        sheet.appendRow(rowData);
      }
      
      return ContentService.createTextOutput(JSON.stringify({ status: "success", action: data.syncAction, staffId: r.staffId }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function recordToArray(r) {
  return [
    r.no || "", r.staffId || "", r.secondaryId || "", r.latinName || "", r.khmerName || "",
    r.department || "", r.office || "", r.position || "", r.gender || "", r.dob || "",
    r.serviceStartDate || "", r.requestDate || "", r.endDate || "", r.startDate || "",
    r.annualPeriod || "", r.requestReason || "", r.prakasNo || "", r.description || "",
    r.systemClosingDate || "", r.refDocument || "", r.receivedDate || "", r.remark || "",
    r.statusLabelKh || "", r.updatedAt || new Date().toLocaleString()
  ];
}
